    <script>

        let apiKey = localStorage.getItem('c2c_auth_key') || "";

        let spamChart;

        window.onload = () => { initCharts(); updateStatus(); };

        function toggleSettings() { document.getElementById('settingsModal').classList.toggle('hidden'); }

        function saveApiKey() { apiKey = document.getElementById('apiKeyInput').value; localStorage.setItem('c2c_auth_key', apiKey); updateStatus(); toggleSettings(); }

        function updateStatus() { const d = document.getElementById('apiStatusDot'); const t = document.getElementById('apiStatusText'); d.className = apiKey ? "w-2 h-2 rounded-full bg-emerald-500" : "w-2 h-2 rounded-full bg-slate-300"; t.innerText = apiKey ? "AI Engine: Ready" : "AI: Offline"; }

        const banks = { 

            subs: [(r)=>`Quick check: ${r} role`, (r)=>`C2C for ${r}?`, (r)=>`Question re: ${r}`],

            feeler: (n,r)=>`Hi ${n},\n\nI noticed you're looking for a ${r}. Are you currently entertaining C2C submissions for this opening?\n\nIf so, I have a strong consultant I can share details on. Please let me know if you're open to it.\n\nBest,\n[Your Name]`,

            pitch: (n,r,s)=>`Thanks ${n}.\n\nAs discussed, here is the resume for our consultant. They have over 10 years experience in ${s} and are available immediately for C2C projects.\n\nLmk if you want to chat!`

        };

        function initCharts() { const ctx = document.getElementById('spamChart').getContext('2d'); spamChart = new Chart(ctx, { type: 'doughnut', data: { labels: ['Safe', 'Risk'], datasets: [{ data: [100, 0], backgroundColor: ['#e2e8f0', '#f1f5f9'], borderWidth: 0 }] }, options: { cutout: '80%', circumference: 180, rotation: 270, plugins: { legend: { display: false } } } }); }

        function predictEmails() {

            const f = document.getElementById('firstName').value.trim().toLowerCase();

            const l = document.getElementById('lastName').value.trim().toLowerCase();

            const d = document.getElementById('domain').value.trim().toLowerCase();

            if(!f || !l || !d) return;

            const res = document.getElementById('emailResults'); res.classList.remove('hidden'); res.innerHTML = `<p class="text-[10px] font-bold text-indigo-600 mb-2">PREDICTIONS</p>`;

            [`${f}.${l}@${d}`, `${f.charAt(0)}${l}@${d}`, `${f}@${d}`].forEach(e => {

                res.innerHTML += `<div class="flex justify-between items-center bg-white p-2 rounded-lg border mb-1 shadow-sm text-[10px] font-mono"><span>${e}</span><button onclick="copy('${e}')" class="text-indigo-600 font-bold">COPY</button></div>`;

            });

        }

        async function auditDomain() {

            const d = document.getElementById('domain').value; if(!d || !apiKey) return alert("Add API key first.");

            try {

                const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, { method: 'POST', body: JSON.stringify({ contents: [{ parts: [{ text: `Email pattern for ${d}` }] }] }) });

                const dat = await r.json(); alert(dat.candidates[0].content.parts[0].text);

            } catch (e) { alert("API Error."); }

        }

        function generate() {

            const r = document.getElementById('roleInput').value || "the role";

            const n = document.getElementById('firstName').value || "there";

            const s = document.getElementById('stackInput').value || "tech";

            const sub = banks.subs[Math.floor(Math.random()*3)](r);

            const m1 = banks.feeler(n,r);

            document.getElementById('sub1').value = sub;

            document.getElementById('msg1').innerText = m1;

            document.getElementById('msg2').innerText = banks.pitch(n,r,s);

            analyze(sub, m1);

        }

        async function humanize() {

            const cur = document.getElementById('msg1').innerText; if(!apiKey || cur.includes("Generating")) return alert("Add API Key and Generate Draft.");

            document.getElementById('aiLoad').classList.remove('hidden');

            try {

                const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, { method: 'POST', body: JSON.stringify({ contents: [{ parts: [{ text: `Rewrite this C2C inquiry to be more human. Max 2 sentences. No bot cliches. Text: ${cur}` }] }] }) });

                const dat = await r.json(); const txt = dat.candidates[0].content.parts[0].text.trim();

                document.getElementById('msg1').innerText = txt; analyze(document.getElementById('sub1').value, txt);

            } catch (e) { alert("AI Error."); }

            document.getElementById('aiLoad').classList.add('hidden');

        }

        function analyze(s, m) {

            const c = (s + m).toLowerCase();

            let score = 60; if (m.length < 250) score += 20; if (c.includes("?")) score += 20; if (c.includes("hope this email finds you well")) score -= 40;

            score = Math.min(Math.max(score, 10), 98);

            document.getElementById('scoreVal').innerText = Math.round(score);

            document.getElementById('humanityGauge').style.background = `radial-gradient(closest-side, white 80%, transparent 81% 100%), conic-gradient(#6366f1 ${score}%, #e2e8f0 0)`;

            document.getElementById('scoreLabel').innerText = score > 80 ? "HUMAN AUTHENTIC" : "BOT PATTERN DETECTED";

            document.getElementById('scoreLabel').className = `text-xs font-bold mt-6 tracking-wider ${score > 80 ? 'text-emerald-500' : 'text-red-500'}`;

            let risk = 10; if (c.includes("urgent")) risk += 30; if (m.length > 500) risk += 20;

            spamChart.data.datasets[0].data = [100 - risk, risk]; 

            spamChart.data.datasets[0].backgroundColor = [risk > 30 ? '#f87171' : '#6366f1', '#f1f5f9'];

            spamChart.update();

        }

        function tab(t) { document.getElementById('step1View').classList.toggle('hidden', t === 'step2'); document.getElementById('step2View').classList.toggle('hidden', t === 'step1'); document.getElementById('t1').classList.toggle('tab-active', t === 'step1'); document.getElementById('t2').classList.toggle('tab-active', t === 'step2'); }

        function copy(i) { const v = document.getElementById(i)?.innerText || document.getElementById(i)?.value || i; navigator.clipboard.writeText(v); }
